			"def alpha_beta(node, alpha, beta, is_maximizing):",
			"    # Base case",
			"    if node['type'] == 'LEAF':",
			"        return node['value']",
			"    ",
			"    print(f\"Visiting {node['name']} | α={alpha} β={beta}\")",
			"    ",
			"    if is_maximizing:",
			"        best = float('-inf')",
			"        for child in node['children']:",
			"            val = alpha_beta(child, alpha, beta, False)",
			"            best = max(best, val)",
			"            alpha = max(alpha, best)          # alpha update karo",
			"            print(f\"  {node['name']} α updated to {alpha}\")",
			"            if alpha >= beta:                 # PRUNE!",
			"                print(f\"  PRUNED at {node['name']}! α({alpha}) >= β({beta})\")",
			"                break",
			"        node['value'] = best",
			"        return best",
			"    else:",
			"        best = float('inf')",
			"        for child in node['children']:",
			"            val = alpha_beta(child, alpha, beta, True)",
			"            best = min(best, val)",
			"            beta = min(beta, best)            # beta update karo",
			"            print(f\"  {node['name']} β updated to {beta}\")",
			"            if alpha >= beta:                 # PRUNE!",
			"                print(f\"  PRUNED at {node['name']}! α({alpha}) >= β({beta})\")",
			"                break",
			"        node['value'] = best",
			"        return best",
			"",
			"result = alpha_beta(tree, float('-inf'), float('inf'), True)",
